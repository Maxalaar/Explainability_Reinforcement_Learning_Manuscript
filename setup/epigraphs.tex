% Chapter epigraphs: optional #2 = English translation (for a quote not
% originally in English), #3 = quote in its original language,
% #4 = attribution (author, work, citation). The starred form omits the
% quote marks around #3, for a quote that brings its own (e.g. Japanese).
\setlength{\epigraphwidth}{0.7\textwidth}
\setlength{\epigraphrule}{0pt}
\renewcommand{\textflush}{flushright}
\renewcommand{\sourceflush}{flushright}
\NewDocumentCommand{\chapterepigraph}{s o m m}{%
  \epigraph{\itshape\IfBooleanTF{#1}{#3}{\enquote{#3}}%
    \IfValueT{#2}{\par\smallskip{\upshape\small\enquote{#2}}}}{#4}%
}
